from decimal import Decimal

from pydantic import TypeAdapter
from pydantic_core import core_schema


class Money:
    def __init__(self, amount, currency):
        self.amount, self.currency = amount, currency

    @classmethod
    def __get_pydantic_core_schema__(cls, source_type, handler):
        mapping_schema = core_schema.typed_dict_schema(
            {
                "amount": core_schema.typed_dict_field(
                    core_schema.str_schema(
                        strict=True,
                        pattern=r"^-?(?:0|[1-9][0-9]*)\.[0-9]{2}$",
                    ),
                    required=True,
                ),
                "currency": core_schema.typed_dict_field(
                    core_schema.literal_schema(["USD", "EUR"]),
                    required=True,
                ),
            },
            extra_behavior="forbid",
        )

        def from_mapping(value):
            return cls(Decimal(value["amount"]), value["currency"])

        def serialize(value):
            return {
                "amount": format(value.amount, ".2f"),
                "currency": value.currency,
            }

        validated_mapping = core_schema.no_info_after_validator_function(
            from_mapping, mapping_schema
        )
        return core_schema.json_or_python_schema(
            json_schema=validated_mapping,
            python_schema=core_schema.union_schema(
                [core_schema.is_instance_schema(cls), validated_mapping]
            ),
            serialization=core_schema.plain_serializer_function_ser_schema(
                serialize, return_schema=mapping_schema
            ),
        )


def make_adapter():
    return TypeAdapter(Money)
